HARNESS (first set; the second set differs only in its lane list)
  #!/bin/bash
  # Parity-gated, interleaved, load-logged benchmark of generate-packages-index.
  # Usage: harness.sh ROUND   (run from the scratch bench dir)
  set -eu
  ROUND=$1
  STAMP=2026-10-14T00:00:00Z
  export PACKAGES_INDEX_GENERATED_AT="$STAMP"
  R=$HOME/dev/hey-lang-bootstrap-plan
  export HEY_ROOT=$R HEY_STDLIB_PATH=$R/stdlib HEY_COMPILER_BUILD_DIR=$R/build
  export PATH="$R/bin:$PATH"          # the shebang's `hey` is 601a, not ~/.config/hey/env.sh's 602a
  OUT=results/r$ROUND; mkdir -p "$OUT"
  
  declare -a NAMES CMDS
  add() { NAMES+=("$1"); CMDS+=("$2"); }
  add ruby407-yjit  "$HOME/.rbenv/versions/4.0.7/bin/ruby --yjit ./generate-packages-index.rb ./packages"
  add ruby407-nojit "env -u RUBY_YJIT_ENABLE $HOME/.rbenv/versions/4.0.7/bin/ruby ./generate-packages-index.rb ./packages"
  add ruby406-yjit  "$HOME/.rbenv/versions/4.0.6/bin/ruby --yjit ./generate-packages-index.rb ./packages"
  add hey-script-w4 './generate-packages-index.hey ./packages'
  add hey-script-w1 './generate-packages-index.hey --workers 1 ./packages'
  add hey-binary-w4 './gpi-bin ./packages'
  add hey-binary-w1 './gpi-bin --workers 1 ./packages'
  add var-script   './variant-stdlib-sort.hey ./packages'
  add var-binary   './variant-bin ./packages'
  
  loadlog() { { echo "== $1 $(date '+%H:%M:%S')"; uptime; ps -Ao pcpu,pid,comm -r | head -6; } >> "$OUT/load.log"; }
  
  # Parity gate: delete, run, assert non-empty, byte-compare against Ruby 4.0.7 + YJIT.
  for i in "${!NAMES[@]}"; do
    rm -f packages/index.html
    bash -c "${CMDS[$i]}" >/dev/null 2>"$OUT/${NAMES[$i]}.stderr"
    test -s packages/index.html || { echo "${NAMES[$i]}: NO OUTPUT"; exit 1; }
    cp packages/index.html "$OUT/out.${NAMES[$i]}.html"
    cmp "$OUT/out.ruby407-yjit.html" "$OUT/out.${NAMES[$i]}.html" || { echo "${NAMES[$i]}: PARITY FAIL"; exit 1; }
  done
  echo "round $ROUND: parity OK for ${#NAMES[@]} lanes"
  
  # Interleave: rotate lane order each round.
  n=${#NAMES[@]}
  for k in $(seq 0 $((n-1))); do
    i=$(( (k + ROUND) % n ))
    loadlog "before ${NAMES[$i]}"
    runs=30; case "${NAMES[$i]}" in ruby*) runs=20;; esac
    nice -n 10 hyperfine -N --warmup 3 --runs $runs --export-json "$OUT/${NAMES[$i]}.json" \
      --command-name "${NAMES[$i]}" "${CMDS[$i]}" >/dev/null
    loadlog "after ${NAMES[$i]}"
    python3 -c "import json,sys;r=json.load(open('$OUT/${NAMES[$i]}.json'))['results'][0];print('%-14s %7.1f ± %5.1f ms  (min %6.1f, n=%d)'%('${NAMES[$i]}',r['mean']*1e3,r['stddev']*1e3,r['min']*1e3,len(r['times'])))"
  done
  rm -f packages/index.html
